     3→     3→/**
     4→     4→ * Zepto-style checkout sheet.
     5→     5→ *
     6→     6→ * A full-height bottom sheet that replaces the legacy inline checkout Dialog
     7→     7→ * in cart-drawer.tsx. The flow is:
     8→     8→ *
     9→     9→ *   Step 1 — "Details"   : Customer name + phone, "Continue to payment"
    10→    10→ *   Step 2 — "Payment"    : Order summary + PaymentMethodPicker + sticky "Pay ₹X" CTA
    11→    11→ *   Step 3 — Method flow : UPI deep-link launch / QR display / mock card+wallet
    12→    12→ *   Step 4 — "Success"   : Confirmation + redirect to tracking
    13→    13→ *
    14→    14→ * The picker is single-select (not "tap = pay"). The customer picks a method,
    15→    15→ * then taps the sticky bottom CTA — exactly like Zepto, Swiggy, Zomato.
    16→    16→ *
    17→    17→ * Used by:
    18→    18→ *   - cart-drawer.tsx (initial order placement)
    19→    19→ *   - bill-view.tsx (pay from the bill view, order already placed)
    20→    20→ *   - order-tracking.tsx (inline "please pay" panel when staff requests payment)
    21→    21→ */

    64→    64→  subtotal: number
    65→    65→  taxAmount: number

    70→    70→interface Props {

    72→    72→  onOpenChange: (open: boolean) => void
    73→    73→  restaurant: RestaurantInfo
    74→    74→  totals: Totals
    75→    75→  items: CartItem[]
    76→    76→  /** Called when an order has been placed + payment (if any) is done. */

    78→    78→  /** Optional: if provided, the sheet operates in "pay existing order" mode

    81→    81→  existingOrderId?: string

    88→    88→   *  post-serve. */
    89→    89→  hiddenMethodIds?: string[]
    90→    90→}
    91→    91→
    92→    92→type Step = 'details' | 'method' | 'upi_launch' | 'qr' | 'processing' | 'success' | 'error'
    93→    93→
    94→    94→const ICON_FOR_METHOD: Record<string, LucideIcon> = {
    95→    95→  UPI: Smartphone,
    96→    96→  QR: QrCode,
    97→    97→  CASH: Banknote,
    98→    98→  CARD: Smartphone,
    99→    99→  WALLET: Smartphone,
   100→   100→  NETBANKING: Smartphone,
   101→   101→  COUNTER: Banknote,
   102→   102→  PAY_LATER: Banknote,
   103→   103→  CUSTOM: Smartphone,

   106→   106→export function CheckoutSheet({
   107→   107→  open,
   108→   108→  onOpenChange,
   109→   109→  restaurant,
   110→   110→  totals,
   111→   111→  items,
   112→   112→  onCheckoutComplete,
   113→   113→  existingOrderId,
   114→   114→  skipCustomerDetails,
   115→   115→  hiddenMethodIds = [],
   116→   116→}: Props) {
   117→   117→  const placeOrder = usePlaceOrder()
   118→   118→  const initiate = useInitiatePayment()
   119→   119→  const verify = useVerifyPayment()
   120→   120→
   121→   121→  const [step, setStep] = useState<Step>(skipCustomerDetails ? 'method' : 'details')
   122→   122→  const [customerName, setCustomerName] = useState('')
   123→   123→  const [customerPhone, setCustomerPhone] = useState('')
   124→   124→  const [touched, setTouched] = useState(false)
   125→   125→  const [selectedMethod, setSelectedMethod] = useState<PaymentMethodT | null>(null)
   126→   126→  const [placedOrderId, setPlacedOrderId] = useState<string | null>(existingOrderId || null)
   127→   127→  const [upiDeepLink, setUpiDeepLink] = useState<string | null>(null)
   128→   128→  const [upiQrPayload, setUpiQrPayload] = useState<string | null>(null)
   129→   129→  const [upiId, setUpiId] = useState<string | null>(null)
   130→   130→  const [successMessage, setSuccessMessage] = useState<string>('')
   131→   131→  const [errorMessage, setErrorMessage] = useState<string>('')
   132→   132→
   133→   133→  // Reset everything when the sheet closes
   134→   134→  useEffect(() => {
   135→   135→    if (!open) {
   136→   136→      const t = setTimeout(() => {
   137→   137→        if (step !== 'processing') {
   138→   138→          setStep(skipCustomerDetails ? 'method' : 'details')
   139→   139→          setSelectedMethod(null)
   140→   140→          setUpiDeepLink(null)
   141→   141→          setUpiQrPayload(null)
   142→   142→          setUpiId(null)
   143→   143→          setTouched(false)
   144→   144→          setSuccessMessage('')
   145→   145→        }
   146→   146→      }, 300)
   147→   147→      return () => clearTimeout(t)
   148→   148→    }
   149→   149→  }, [open, step, skipCustomerDetails])
   150→   150→
   151→   151→  const availableMethods = useMemo(
   152→   152→    () => filterAvailableForCheckout(restaurant.paymentMethods),
   153→   153→    [restaurant.paymentMethods],
   154→   154→  )
   155→   155→
   156→   156→  const nameValid = customerName.trim().length >= 2
   157→   157→  const digits = customerPhone.replace(/[^\d]/g, '')
   158→   158→  const phoneValid = digits.length >= 7 && digits.length <= 15
   159→   159→  const formValid = nameValid && phoneValid
   160→   160→
   161→   161→  const grandTotal = totals.grandTotal
   162→   162→  const totalDisplay = `₹${grandTotal.toFixed(0)}`
   163→   163→
   164→   164→  // --------------------------------------------------------------------------
   165→   165→  // Action: place order (only when no existingOrderId was provided)
   166→   166→  // --------------------------------------------------------------------------
   167→   167→  async function ensureOrderPlaced(): Promise<string | null> {
   168→   168→    if (placedOrderId) return placedOrderId
   169→   169→    if (!formValid) {
   170→   170→      setTouched(true)
   171→   171→      toast.error('Please enter your name and phone number.')
   172→   172→      return null
   173→   173→    }
   174→   174→    const idempotencyKey =
   175→   175→      sessionStorage.getItem('last-idem-key') ||
   176→   176→      `${Date.now()}-${Math.random().toString(36).slice(2, 14)}`
   177→   177→    sessionStorage.setItem('last-idem-key', idempotencyKey)
   178→   178→
   179→   179→    const body = {
   180→   180→      tableToken: new URLSearchParams(window.location.search).get('table') || '',
   181→   181→      items: items.map((i: CartItem) => ({
   182→   182→        menuItemId: i.menuItemId,
   183→   183→        variantId: i.variantId,
   184→   184→        modifierIds: i.modifierIds,
   185→   185→        quantity: i.quantity,
   186→   186→        notes: i.notes,
   187→   187→      })),
   188→   188→      idempotencyKey,
   189→   189→      customerInfo: {
   190→   190→        name: customerName.trim(),
   191→   191→        phone: customerPhone.trim(),
   192→   192→      },
   193→   193→      // Persist the chosen payment-method ID on the order so the post-serve
   194→   194→      // picker can hide the same pay-on-delivery method if the customer picked
   195→   195→      // one (Cash / Counter / Pay Later).
   196→   196→      paymentMethodId: selectedMethod?.id || null,
   197→   197→    }
   198→   198→    const order = await placeOrder.mutateAsync(body)
   199→   199→    sessionStorage.removeItem('last-idem-key')
   200→   200→    setPlacedOrderId(order.id)
   201→   201→    return order.id
   202→   202→  }
   203→   203→
   204→   204→  // --------------------------------------------------------------------------
   205→   205→  // Action: trigger the selected payment method's flow
   206→   206→  // --------------------------------------------------------------------------
   207→   207→  async function handlePay() {
   208→   208→    if (!selectedMethod) {
   209→   209→      toast.error('Please choose a payment method.')
   210→   210→      return
   211→   211→    }
   212→   212→
   213→   213→    try {
   214→   214→      const orderId = await ensureOrderPlaced()
   215→   215→      if (!orderId) return
   216→   216→
   217→   217→      const t = selectedMethod.type
   218→   218→      // Cash / Counter / Pay-Later: no online payment — staff mark as paid.
   219→   219→      if (t === 'CASH' || t === 'COUNTER' || t === 'PAY_LATER') {
   220→   220→        setStep('processing')
   221→   221→        // Brief processing delay for UX, then show success
   222→   222→        await new Promise((r) => setTimeout(r, 600))
   223→   223→        setSuccessMessage(
   224→   224→          t === 'CASH'
   225→   225→            ? 'Order placed! Please hand the cash to your waiter.'
   226→   226→            : t === 'COUNTER'
   227→   227→              ? 'Order placed! Please pay at the billing counter.'
   228→   228→              : 'Order placed! You can settle the bill before leaving.',
   229→   229→        )
   230→   230→        setStep('success')
   231→   231→        toast.success(successMessage, { duration: 4000 })
   232→   232→        return
   233→   233→      }
   234→   234→
   235→   235→      // Online flow (UPI / QR / CARD / WALLET / NETBANKING / CUSTOM)
   236→   236→      setStep('processing')
   237→   237→      const apiMethod = t === 'QR' ? 'UPI' : t
   238→   238→      const init = await initiate.mutateAsync({
   239→   239→        orderId,
   240→   240→        method: apiMethod as any,
   241→   241→        paymentMethodId: selectedMethod.id,
   242→   242→      })
   243→   243→
   244→   244→      setUpiDeepLink(init.upiDeepLink || null)
   245→   245→      setUpiQrPayload(init.upiQrPayload || null)
   246→   246→      setUpiId(init.upiId || null)
   247→   247→
   248→   248→      // UPI: open the deep-link (auto-launch customer's UPI app), then wait
   249→   249→      // for them to return + verify.
   250→   250→      if (t === 'UPI' && init.upiDeepLink) {
   251→   251→        const tableToken = new URLSearchParams(window.location.search).get('table') || ''
   252→   252→        sessionStorage.setItem(`order-${tableToken}`, orderId)
   253→   253→        sessionStorage.setItem('returning-from-upi', '1')
   254→   254→        try {
   255→   255→          window.open(init.upiDeepLink, '_blank')
   256→   256→        } catch {
   257→   257→          // popup blocked — fall through to the manual "Open UPI app" button
   258→   258→        }
   259→   259→        setStep('upi_launch')
   260→   260→        await new Promise((r) => setTimeout(r, init.verifyInMs || 3000))
   261→   261→        await verify.mutateAsync({
   262→   262→          paymentId: init.paymentId,
   263→   263→          providerTxnId: init.providerTxnId,
   264→   264→        })
   265→   265→        setSuccessMessage('Payment successful! Your order is confirmed.')
   266→   266→        setStep('success')
   267→   267→        return
   268→   268→      }
   269→   269→
   270→   270→      // QR: show the QR + UPI ID, wait for the customer to scan + pay, then verify
   271→   271→      if (t === 'QR' && init.upiQrPayload) {
   272→   272→        setStep('qr')
   273→   273→        await new Promise((r) => setTimeout(r, init.verifyInMs || 3000))
   274→   274→        await verify.mutateAsync({
   275→   275→          paymentId: init.paymentId,
   276→   276→          providerTxnId: init.providerTxnId,
   277→   277→        })
   278→   278→        setSuccessMessage('Payment successful! Your order is confirmed.')
   279→   279→        setStep('success')
   280→   280→        return
   281→   281→      }
   282→   282→
   283→   283→      // CARD / WALLET / NETBANKING / CUSTOM: mock wait, then verify
   284→   284→      await new Promise((r) => setTimeout(r, init.verifyInMs || 1500))
   285→   285→      await verify.mutateAsync({
   286→   286→        paymentId: init.paymentId,
   287→   287→        providerTxnId: init.providerTxnId,
   288→   288→      })
   289→   289→      setSuccessMessage('Payment successful! Your order is confirmed.')
   290→   290→      setStep('success')
   291→   291→    } catch (err: any) {
   292→   292→      // CRITICAL: do NOT silently reset to 'method'. If the order was already
   293→   293→      // placed, the customer has a placed order in the system — we MUST route
   294→   294→      // them to the tracking page (or give them an explicit way to retry).
   295→   295→      //
   296→   296→      // - If placedOrderId is set (order was created in this sheet OR the sheet
   297→   297→      //   was opened in existing-order mode) → go to 'error' step which offers
   298→   298→      //   "Track order" + "Try again" so they're never stuck.
   299→   299→      // - Otherwise (e.g. placeOrder itself failed) → back to 'method' so they
   300→   300→      //   can retry selecting a method, but still surface the error.
   301→   301→      const msg = err?.message || 'Payment failed. Please try again.'
   302→   302→      setErrorMessage(msg)
   303→   303→      if (placedOrderId) {
   304→   304→        setStep('error')
   305→   305→      } else {
   306→   306→        setStep('method')
   307→   307→      }
   308→   308→      toast.error(msg)
   309→   309→    }
   310→   310→  }
   311→   311→
   312→   312→  // Auto-redirect to order tracking after success
   313→   313→  useEffect(() => {
   314→   314→    if (step === 'success' && placedOrderId) {
   315→   315→      const t = setTimeout(() => {
   316→   316→        onCheckoutComplete(placedOrderId)
   317→   317→      }, 1800)
   318→   318→      return () => clearTimeout(t)
   319→   319→    }
   320→   320→  }, [step, placedOrderId, onCheckoutComplete])
   321→   321→
   322→   322→  // Manual redirect from the error step → route to tracking page
   323→   323→  const goToTracking = () => {
   324→   324→    if (placedOrderId) {
   325→   325→      onCheckoutComplete(placedOrderId)
   326→   326→    }
   327→   327→  }
   328→   328→
   329→   329→  // Retry payment on the SAME order (clears the failed attempt and lets the
   330→   330→  // customer pick a different method or retry the same one). We keep
   331→   331→  // placedOrderId intact so `ensureOrderPlaced()` returns the existing order.
   332→   332→  const retryPayment = () => {
   333→   333→    setErrorMessage('')
   334→   334→    setSelectedMethod(null)
   335→   335→    setUpiDeepLink(null)
   336→   336→    setUpiQrPayload(null)
   337→   337→    setUpiId(null)
   338→   338→    setStep('method')
   339→   339→  }
   340→   340→
   341→   341→  const canPay = !!selectedMethod && (skipCustomerDetails || formValid) && !placeOrder.isPending
   342→   342→
   343→   343→  // --------------------------------------------------------------------------
   344→   344→  // Render
   345→   345→  // --------------------------------------------------------------------------
   346→   346→  return (
   347→   347→    <Sheet
   348→   348→      open={open}
   349→   349→      onOpenChange={(o) => {
   350→   350→        // Block close during processing + success (auto-redirect in flight)
   351→   351→        if (step === 'processing' || step === 'success') return
   352→   352→        // If the customer tries to close the sheet during the 'error' step
   353→   353→        // AND the order was placed, route them to the tracking page instead
   354→   354→        // of just dismissing — they have an order in the system that they
   355→   355→        // need to be able to track. Only fall through to a normal close if
   356→   356→        // there's no placed order (e.g. the initial placeOrder call itself
   357→   357→        // failed, so there's nothing to track).
   358→   358→        if (!o && step === 'error' && placedOrderId) {
   359→   359→          onCheckoutComplete(placedOrderId)
   360→   360→          return
   361→   361→        }
   362→   362→        onOpenChange(o)
   363→   363→      }}
   364→   364→    >
   365→   365→      <SheetContent
   366→   366→        side="bottom"
   367→   367→        className="flex max-h-[94vh] flex-col gap-0 rounded-t-3xl border-slate-200 bg-white p-0"
   368→   368→      >
   369→   369→        {/* Drag handle */}
   370→   370→        <div className="mx-auto mt-2 h-1 w-10 rounded-full bg-slate-300" />
   371→   371→
   372→   372→        {/* Header — varies by step */}
   373→   373→        <SheetHeader className="border-b border-slate-100 px-5 pb-3 pt-3">
   374→   374→          <div className="flex items-center gap-2">
   375→   375→            {step === 'method' && !skipCustomerDetails && (
   376→   376→              <button
   377→   377→                onClick={() => setStep('details')}
   378→   378→                className="rounded-full p-1 text-slate-500 hover:bg-slate-100"
   379→   379→                aria-label="Back"
   380→   380→              >
   381→   381→                <ChevronLeft className="h-5 w-5" />
   382→   382→              </button>
   383→   383→            )}
   384→   384→            <SheetTitle className="text-left text-[16px] font-bold text-slate-900">
   385→   385→              {stepTitle(step)}
   386→   386→            </SheetTitle>
   387→   387→          </div>
   388→   388→          <SheetDescription className="sr-only">
   389→   389→            {stepDescription(step)}
   390→   390→          </SheetDescription>
   391→   391→        </SheetHeader>
   392→   392→
   393→   393→        {/* Body — scrollable */}
   394→   394→        <div className="flex-1 overflow-y-auto px-5 py-4">
   395→   395→          <AnimatePresence mode="wait">
   396→   396→            {/* Step 1: Customer details */}
   397→   397→            {step === 'details' && (
   398→   398→              <motion.div
   399→   399→                key="details"
   400→   400→                initial={{ opacity: 0, x: 8 }}
   401→   401→                animate={{ opacity: 1, x: 0 }}
   402→   402→                exit={{ opacity: 0, x: -8 }}
   403→   403→                className="space-y-4"
   404→   404→              >
   405→   405→                <div className="rounded-xl bg-slate-50 p-3">
   406→   406→                  <div className="flex items-center justify-between text-[13px]">
   407→   407→                    <span className="text-slate-500">Paying to</span>
   408→   408→                    <span className="font-semibold text-slate-800">{restaurant.name}</span>
   409→   409→                  </div>
   410→   410→                  <div className="mt-1 flex items-center justify-between text-[13px]">
   411→   411→                    <span className="text-slate-500">Amount</span>
   412→   412→                    <span className="text-lg font-extrabold text-slate-900">{totalDisplay}</span>
   413→   413→                  </div>
   414→   414→                </div>
   415→   415→
   416→   416→                <div className="space-y-3">
   417→   417→                  <div className="space-y-1.5">
   418→   418→                    <Label htmlFor="cs-name" className="text-[13px] font-semibold text-slate-700">
   419→   419→                      Your name <span className="text-red-500">*</span>
   420→   420→                    </Label>
   421→   421→                    <Input
   422→   422→                      id="cs-name"
   423→   423→                      placeholder="e.g. Arjun Patel"
   424→   424→                      value={customerName}
   425→   425→                      onChange={(e) => setCustomerName(e.target.value)}
   426→   426→                      maxLength={80}
   427→   427→                      autoFocus
   428→   428→                      aria-invalid={touched && !nameValid}
   429→   429→                      className="h-11 rounded-xl border-slate-200 text-[14px] focus-visible:ring-orange-400"
   430→   430→                    />
   431→   431→                    {touched && !nameValid && (
   432→   432→                      <p className="text-[11px] text-red-500">Please enter your name (min 2 characters).</p>
   433→   433→                    )}
   434→   434→                  </div>
   435→   435→
   436→   436→                  <div className="space-y-1.5">
   437→   437→                    <Label htmlFor="cs-phone" className="text-[13px] font-semibold text-slate-700">
   438→   438→                      Phone number <span className="text-red-500">*</span>
   439→   439→                    </Label>
   440→   440→                    <Input
   441→   441→                      id="cs-phone"
   442→   442→                      inputMode="tel"
   443→   443→                      placeholder="e.g. +91 98765 43210"
   444→   444→                      value={customerPhone}
   445→   445→                      onChange={(e) => setCustomerPhone(e.target.value)}
   446→   446→                      maxLength={20}
   447→   447→                      aria-invalid={touched && !phoneValid}
   448→   448→                      className="h-11 rounded-xl border-slate-200 text-[14px] focus-visible:ring-orange-400"
   449→   449→                    />
   450→   450→                    {touched && !phoneValid && (
   451→   451→                      <p className="text-[11px] text-red-500">Please enter a valid phone number (7–15 digits).</p>
   452→   452→                    )}
   453→   453→                  </div>
   454→   454→                </div>
   455→   455→
   456→   456→                <div className="flex items-center gap-2 rounded-lg bg-emerald-50 p-2.5 text-[11px] text-emerald-700">
   457→   457→                  <ShieldCheck className="h-4 w-4 shrink-0" />
   458→   458→                  <span>Your details are shared only with the restaurant to fulfil your order.</span>
   459→   459→                </div>
   460→   460→              </motion.div>
   461→   461→            )}
   462→   462→
   463→   463→            {/* Step 2: Payment method picker */}
   464→   464→            {step === 'method' && (
   465→   465→              <motion.div

   815→   815→/** Render a QR data URL via the lazy-loaded qrcode lib. */
   816→   816→function QrCodeDisplay({ payload }: { payload: string }) {
   817→   817→  const [dataUrl, setDataUrl] = useState<string | null>(null)
   818→   818→  const [error, setError] = useState(false)
   819→   819→
   820→   820→  useEffect(() => {
   821→   821→    let cancelled = false
   822→   822→    import('qrcode')
   823→   823→      .then((QRCode) =>
   824→   824→        QRCode.toDataURL(payload, {
   825→   825→          width: 256,
   826→   826→          margin: 1,
   827→   827→          color: { dark: '#0f172a', light: '#ffffff' },
   828→   828→        }),
   829→   829→      )
   830→   830→      .then((url: string) => {
   831→   831→        if (!cancelled) setDataUrl(url)
   832→   832→      })
   833→   833→      .catch(() => {
   834→   834→        if (!cancelled) setError(true)
   835→   835→      })
   836→   836→      .catch(() => {
   837→   837→        if (!cancelled) setError(true)
   838→   838→      })
   839→   839→    return () => {
   840→   840→      cancelled = true
   841→   841→    }
   842→   842→  }, [payload])
   843→   843→
   844→   844→  if (error) {
   845→   845→    return (
   846→   846→      <div className="flex h-48 w-48 items-center justify-center text-center text-xs text-red-500">
   847→   847→        Failed to generate QR code.
   848→   848→        <br />
   849→   849→        Please use the UPI ID directly.
   850→   850→      </div>
   851→   851→    )
   852→   852→  }
   853→   853→  if (!dataUrl) {
   854→   854→    return (
   855→   855→      <div className="flex h-48 w-48 items-center justify-center">
   856→   856→        <Loader2 className="h-6 w-6 animate-spin text-slate-300" />
   857→   857→      </div>
   858→   858→    )
   859→   859→  }
   860→   860→  return <img src={dataUrl} alt="UPI payment QR code" className="h-48 w-48" />
   861→   861→}
   862→   862→
