    55→// ---------------------------------------------------------------------------
    56→// Status config — drives the hero card + stepper
    57→// ---------------------------------------------------------------------------
    58→interface StepConfig {
    59→  key: OrderStatus
    60→  label: string
    61→  shortLabel: string
    62→  sublabel: string
    63→  icon: LucideIcon
    64→  etaText?: string
    65→}
    66→
    67→const STEPS: StepConfig[] = [
    68→  { key: 'NEW', label: 'Order Placed', shortLabel: 'Placed', sublabel: 'Waiting for restaurant to accept', icon: CheckCircle2, etaText: '~20-25 min' },
    69→  { key: 'ACCEPTED', label: 'Confirmed', shortLabel: 'Confirmed', sublabel: 'Restaurant has accepted your order', icon: BellRing, etaText: '~15-20 min' },
    70→  { key: 'PREPARING', label: 'Preparing', shortLabel: 'Cooking', sublabel: 'The chef is preparing your food', icon: ChefHat, etaText: '~10-15 min' },
    71→  { key: 'READY', label: 'Ready to Serve', shortLabel: 'Ready', sublabel: 'Your food is ready for serving', icon: PackageCheck, etaText: '~2-5 min' },
    72→  { key: 'SERVED', label: 'Served', shortLabel: 'Served', sublabel: 'Enjoy your meal!', icon: Utensils, etaText: 'Enjoy!' },
    73→]
    74→
    75→function stepIndex(status: string): number {
    76→  const order: string[] = ['NEW', 'ACCEPTED', 'PREPARING', 'READY', 'SERVED', 'COMPLETED']
    77→  const i = order.indexOf(status)
    78→  if (i < 0) return -1
    79→  return Math.min(i, STEPS.length - 1)
    80→}

    85→export function OrderTracking({
    86→  orderId,
    87→  restaurant,
    88→  onBackToMenu,
    89→  onProceedToBill,
    90→}: {
    91→  orderId: string
    92→  restaurant?: RestaurantInfo
    93→  onBackToMenu: () => void
    94→  onProceedToBill: () => void
    95→}) {
    96→  const qc = useQueryClient()
    97→  const { data: order, isLoading } = useCustomerOrder(orderId)
    98→  const cancel = useCancelOrder()
    99→  const [checkoutOpen, setCheckoutOpen] = useState(false)
   100→  const [postPayPrompted, setPostPayPrompted] = useState(false)
   101→  const [billExpanded, setBillExpanded] = useState(false)
   102→
   103→  // ---- Socket subscriptions (always called, regardless of loading state) ----
   104→  useSocketEvent('order:updated', (payload: any) => {
   105→    if (payload?.orderId === orderId) {
   106→      qc.invalidateQueries({ queryKey: ['customer-order', orderId] })
   107→    }
   108→  })
   109→  useSocketEvent('order:statusChanged', (payload: any) => {
   110→    if (payload?.orderId === orderId) {
   111→      qc.invalidateQueries({ queryKey: ['customer-order', orderId] })
   112→      toast.success(`Order status: ${payload.status}`)
   113→    }
   114→  })
   115→  useSocketEvent('payment:confirmed', (payload: any) => {
   116→    if (payload?.orderId === orderId) {
   117→      qc.invalidateQueries({ queryKey: ['customer-order', orderId] })
   118→      toast.success('Payment confirmed!')
   119→    }
   120→  })
   121→  useSocketEvent('payment:requested', (payload: any) => {
   122→    if (payload?.orderId === orderId) {
   123→      qc.invalidateQueries({ queryKey: ['customer-order', orderId] })
   124→      const when = payload.when === 'PRE' ? 'before we accept your order' : 'now that your order is received'
   125→      toast.info(`Payment requested: please pay ${when}.`, {
   126→        description: `Amount: ₹${(payload.amount || 0).toFixed(0)}`,
   127→      })
   128→    }
   129→  })
   130→
   131→  // ---- Derived state (safe to compute even if order is undefined) ----
   132→  const isPaid = order?.paymentStatus === 'PAID'
   133→  const isCompleted = order?.status === 'COMPLETED'
   134→  const isPendingPayment = order?.status === 'PENDING_PAYMENT'
   135→  const isServed = order?.status === 'SERVED'
   136→  const needsToPay =
   137→    !!order &&
   138→    !isPaid &&
   139→    (isPendingPayment || !!order.prePaymentRequested || !!order.postPaymentRequested)
   140→
   141→  // ---- Auto-open the checkout sheet once when order is SERVED ----
   142→  // (Zepto-style post-meal payment prompt). MUST be before any early return.
   143→  useEffect(() => {
   144→    if (
   145→      isServed &&
   146→      !isPaid &&
   147→      !postPayPrompted &&
   148→      !sessionStorage.getItem(`postpay-popup-${orderId}`)
   149→    ) {
   150→      setCheckoutOpen(true)
   151→      sessionStorage.setItem(`postpay-popup-${orderId}`, 'shown')
   152→      setPostPayPrompted(true)
   153→    }
   154→  }, [isServed, isPaid, orderId, postPayPrompted])
   155→
   156→  // ---- Loading state (only AFTER all hooks are declared) ----
   157→  if (isLoading || !order) {
   158→    return (
   159→      <div className="flex min-h-[70vh] items-center justify-center bg-gradient-to-b from-orange-50/40 to-white">
   160→        <div className="text-center">
   161→          <div className="mx-auto mb-3 flex h-14 w-14 items-center justify-center rounded-2xl bg-orange-50">
   162→            <ChefHat className="h-7 w-7 text-orange-500 animate-pulse" />
   163→          </div>
   164→          <p className="text-[14px] font-semibold text-slate-700">Loading your order…</p>
   165→          <p className="mt-1 text-[12px] text-slate-400">Just a moment</p>
   166→        </div>
   167→      </div>
   168→    )
   169→  }
   170→
   171→  // ---- Now safe to use order (loaded) ----
   172→  const curStep = stepIndex(order.status)
   173→  const canCancel = order.status === 'NEW' || order.status === 'PENDING_PAYMENT'
   174→  const canRequestBill = ['SERVED', 'READY'].includes(order.status) && order.paymentStatus !== 'PAID'
   175→  const currentStepConfig = STEPS[Math.max(curStep, 0)]
   176→  const etaText = isCompleted ? null : currentStepConfig?.etaText
   177→  const HeroIcon = currentStepConfig?.icon || Clock
   178→  const itemCount = order.items?.length || 0
   179→
   180→  // Compute which payment-method IDs to hide from the post-serve picker.
   181→  //
   182→  // Rule: when the customer already chose a pay-on-delivery method at order
   183→  // placement, hide Counter + Pay Later (those mean "pay at counter / settle
   184→  // later" — they'd be confusing to show again post-serve). KEEP Cash visible
   185→  // because the waiter can still collect cash at the table — and keep all the
   186→  // online methods (UPI / QR / Card / Wallet / Net Banking) so the customer
   187→  // can switch to paying online if they prefer.
   188→  //
   189→  // Net effect on the post-serve picker when customer chose pay-on-delivery:
   190→  //   ✅ UPI   ✅ Scan QR   ✅ Card   ✅ Wallet   ✅ Net Banking   ✅ Cash
   191→  //   ❌ Pay at Counter   ❌ Pay Later
   192→  const hiddenMethodIds: string[] = (() => {
   193→    const chosenId = (order as any)?.paymentMethodId as string | null | undefined
   194→    const chosenType = (order as any)?.paymentMethod as string | null | undefined
   195→    if (!chosenId || !chosenType) return []
   196→    if (!PAY_ON_DELIVERY_TYPES.includes(chosenType)) return []
   197→    // Customer chose a pay-on-delivery method → hide only COUNTER + PAY_LATER
   198→    // (Cash stays visible — waiter can still collect cash at the table).
   199→    return (restaurant?.paymentMethods || [])
   200→      .filter((m) => HIDE_POST_SERVE_TYPES.includes(m.type))
   201→      .map((m) => m.id)
   202→  })()
   203→
   204→  return (
   205→    <div className="min-h-screen bg-gradient-to-b from-orange-50/40 via-white to-white pb-44 sm:pb-28">
   206→      {/* ----- Sticky header ----- */}
   207→      <div className="sticky top-0 z-20 border-b border-orange-100/60 bg-white/80 backdrop-blur-md">
   208→        <div className="mx-auto flex max-w-2xl items-center gap-3 px-4 py-3">
   209→          <button
   210→            onClick={onBackToMenu}
   211→            className="-ml-1 flex h-9 w-9 items-center justify-center rounded-full text-slate-500 transition-colors hover:bg-slate-100 hover:text-slate-700"
   212→            aria-label="Back to menu"
   213→          >
   214→            <ArrowLeft className="h-5 w-5" />
   215→          </button>
   216→          <div className="flex-1">
   217→            <div className="flex items-center gap-2">
   218→              <h1 className="text-[16px] font-extrabold tracking-tight text-slate-900">
   219→                {order.orderNumber}
   220→              </h1>
   221→              <OrderStatusBadge status={order.status} />
   222→            </div>
   223→            <div className="mt-0.5 flex items-center gap-1.5 text-[11.5px] text-slate-400">
   224→              <MapPin className="h-3 w-3" />
   225→              <span>Table {order.table?.number} · Dine-in</span>
   226→              <span className="mx-0.5">·</span>
   227→              <PaymentStatusBadge status={order.paymentStatus} />
   228→            </div>
   229→          </div>
   230→        </div>
   231→      </div>
   232→
   233→      <div className="mx-auto max-w-2xl px-4 pt-4">
   234→        {/* ----- Hero status card (Zepto-style big card) ----- */}
   235→        <motion.div
   236→          initial={{ y: 12, opacity: 0 }}
   237→          animate={{ y: 0, opacity: 1 }}
   238→          transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
   239→          className="relative overflow-hidden rounded-3xl border border-orange-100 bg-gradient-to-br from-orange-50 via-white to-orange-50/30 p-5 shadow-sm"
   240→        >
   241→          {/* Decorative glow */}
   242→          <div className="pointer-events-none absolute -right-12 -top-12 h-40 w-40 rounded-full bg-orange-200/30 blur-3xl" />
   243→
   244→          <div className="relative flex items-start justify-between">
   245→            <div className="flex-1">
   246→              <p className="text-[12px] font-bold uppercase tracking-wider text-orange-600">
   247→                {isCompleted ? 'Order Complete' : 'Status'}
   248→              </p>
   249→              <h2 className="mt-1 text-[24px] font-extrabold leading-tight text-slate-900">
   250→                {isCompleted ? 'All done!' : currentStepConfig?.label || 'Processing'}
   251→              </h2>
   252→              <p className="mt-1 text-[13px] text-slate-500">
   253→                {currentStepConfig?.sublabel}
   254→              </p>
   255→
   256→              {etaText && !isCompleted && (
   257→                <div className="mt-3 inline-flex items-center gap-1.5 rounded-full bg-white px-3 py-1.5 text-[12px] font-semibold text-orange-700 shadow-sm ring-1 ring-orange-100">
   258→                  <Clock className="h-3.5 w-3.5" />
   259→                  ETA {etaText}
   260→                </div>
   261→              )}
   262→            </div>
   263→
   264→            {/* Animated status icon */}
   265→            <motion.div
   266→              initial={{ scale: 0.8, rotate: -5 }}
   267→              animate={{ scale: 1, rotate: 0 }}
   268→              transition={{ type: 'spring', stiffness: 300, damping: 18, delay: 0.1 }}
   269→              className={cn(
   270→                'flex h-16 w-16 shrink-0 items-center justify-center rounded-2xl shadow-md',
   271→                isCompleted ? 'bg-green-100' : 'bg-orange-500',
   272→              )}
   273→            >
   274→              <HeroIcon
   275→                className={cn('h-8 w-8', isCompleted ? 'text-green-600' : 'text-white')}
   276→                strokeWidth={2.25}
   277→              />
   278→            </motion.div>
   279→          </div>
   280→        </motion.div>
   281→
   282→        {/* ----- Horizontal progress stepper (Zepto-style) ----- */}
   283→        <motion.div
   284→          initial={{ y: 12, opacity: 0 }}
   285→          animate={{ y: 0, opacity: 1 }}
   286→          transition={{ duration: 0.4, delay: 0.08, ease: [0.22, 1, 0.36, 1] }}
   287→          className="mt-4 rounded-2xl border border-slate-100 bg-white p-4 shadow-sm"
   288→        >
   289→          <div className="flex items-center justify-between">
   290→            {STEPS.map((step, idx) => {
   291→              const Icon = step.icon
   292→              const isActive = idx === curStep
   293→              const isDone = idx < curStep
   294→              const isLast = idx === STEPS.length - 1
   295→
   296→              return (
   297→                <div key={step.key} className="flex flex-1 items-center">
   298→                  {/* Step circle */}
   299→                  <div className="flex flex-col items-center gap-1">
   300→                    <motion.div
   301→                      initial={false}
   302→                      animate={{
   303→                        scale: isActive ? 1.12 : 1,
   304→                        backgroundColor: isDone || isActive ? '#ea580c' : '#f1f5f9',
   305→                      }}
   306→                      transition={{ type: 'spring', stiffness: 400, damping: 25 }}
   307→                      className={cn(
   308→                        'flex h-9 w-9 items-center justify-center rounded-full ring-2 transition-colors',
   309→                        isActive ? 'ring-orange-200' : 'ring-transparent',
   310→                      )}
   311→                    >
   312→                      {isDone ? (
   313→                        <CheckCircle2 className="h-5 w-5 text-white" strokeWidth={2.5} />
   314→                      ) : (
   315→                        <Icon
   316→                          className={cn(
   317→                            'h-4 w-4',
   318→                            isActive ? 'text-white' : 'text-slate-400',
   319→                          )}
   320→                          strokeWidth={isActive ? 2.5 : 1.75}
   321→                        />
   322→                      )}
   323→                    </motion.div>
   324→                    <span
   325→                      className={cn(
   326→                        'text-[10px] font-semibold tracking-wide',
   327→                        isActive ? 'text-orange-600' : isDone ? 'text-slate-700' : 'text-slate-400',
   328→                      )}
   329→                    >
   330→                      {step.shortLabel}
   331→                    </span>
   332→                  </div>
   333→
   334→                  {/* Connecting line */}
   335→                  {!isLast && (
   336→                    <div className="mx-1 mb-5 h-1 flex-1 overflow-hidden rounded-full bg-slate-100">
   337→                      <motion.div
   338→                        initial={false}
   339→                        animate={{ width: isDone ? '100%' : '0%' }}
   340→                        transition={{ duration: 0.5, ease: 'easeInOut' }}
   341→                        className="h-full bg-gradient-to-r from-orange-400 to-orange-500"
   342→                      />
   343→                    </div>
   344→                  )}
   345→                </div>
   346→              )
   347→            })}
   348→          </div>
   349→        </motion.div>
   350→
   351→        {/* ----- Payment-CTA card (Zepto-style — shows when payment is due) ----- */}
   352→        <AnimatePresence>
   353→          {needsToPay && (
   354→            <motion.div
   355→              initial={{ y: 12, opacity: 0, height: 0 }}
   356→              animate={{ y: 0, opacity: 1, height: 'auto' }}
   357→              exit={{ y: 8, opacity: 0, height: 0 }}
   358→              transition={{ duration: 0.3 }}
   359→              className="mt-4 overflow-hidden rounded-2xl border border-amber-200 bg-gradient-to-br from-amber-50 to-orange-50 shadow-sm"
   360→            >
   361→              <div className="p-4">
   362→                <div className="mb-1 flex items-center gap-2">
   363→                  <div className="flex h-7 w-7 items-center justify-center rounded-lg bg-amber-100">
   364→                    <CreditCard className="h-4 w-4 text-amber-700" />
   365→                  </div>
   366→                  <h3 className="text-[14px] font-bold text-amber-900">
   367→                    {isPendingPayment || order.prePaymentRequested
   368→                      ? 'Please pay to confirm'
   369→                      : 'Settle your bill'}
   370→                  </h3>
   371→                </div>
   372→                <p className="mb-3 text-[12.5px] text-amber-700/90">
   373→                  {isPendingPayment || order.prePaymentRequested
   374→                    ? 'Your order goes to the kitchen once payment is confirmed.'
   375→                    : 'Please settle the bill before leaving.'}
   376→                </p>
   377→                <div className="flex items-center justify-between rounded-xl bg-white p-3 shadow-sm">
   378→                  <span className="text-[13px] font-medium text-slate-600">Amount due</span>
   379→                  <span className="text-[18px] font-extrabold text-slate-900">
   380→                    {formatINR(order.grandTotal)}
   381→                  </span>
   382→                </div>
   383→              </div>
   384→            </motion.div>
   385→          )}
   386→        </AnimatePresence>
   387→
   388→        {/* ----- Order items card ----- */}
   389→        <motion.div
   390→          initial={{ y: 12, opacity: 0 }}
   391→          animate={{ y: 0, opacity: 1 }}
   392→          transition={{ duration: 0.4, delay: 0.16, ease: [0.22, 1, 0.36, 1] }}
   393→          className="mt-4 overflow-hidden rounded-2xl border border-slate-100 bg-white shadow-sm"
   394→        >
   395→          <div className="flex items-center justify-between border-b border-slate-100 px-4 py-3">
   396→            <h3 className="text-[14px] font-bold text-slate-800">
   397→              Your order
   398→              <span className="ml-1.5 text-[12px] font-medium text-slate-400">
   399→                ({itemCount} item{itemCount !== 1 ? 's' : ''})
   400→              </span>
   401→            </h3>
   402→            <span className="text-[13px] font-bold text-slate-700">
   403→              {formatINR(order.grandTotal)}
   404→            </span>
   405→          </div>
   406→
   407→          <div className="divide-y divide-slate-50">
   408→            {order.items?.map((it: any) => (
   409→              <div key={it.id} className="flex items-start gap-2.5 px-4 py-3">
   410→                <div className="mt-0.5">
   411→                  <VegBadge isVeg={it.isVeg} />
   412→                </div>
   413→                <div className="min-w-0 flex-1">
   414→                  <p className="text-[13.5px] font-semibold text-slate-800">

   416→                  </p>

   418→                    <p className="mt-0.5 text-[11.5px] text-slate-400">{it.variantName}</p>
   419→                  )}

   421→                    <p className="mt-0.5 truncate text-[11.5px] text-slate-400">

   430→                  <p className="text-[13.5px] font-bold text-slate-900">

   434→              </div>

   437→
   438→          {/* Bill details — collapsible */}
   439→          <button

   441→            className="flex w-full items-center justify-between border-t border-slate-100 px-4 py-3 text-[13px] font-semibold text-slate-700 transition-colors hover:bg-slate-50"
   442→          >
   443→            <span className="flex items-center gap-2">
   444→              <Receipt className="h-4 w-4 text-slate-400" />
   445→              Bill details
   446→            </span>
   447→            <ChevronDown
   448→              className={cn(
   449→                'h-4 w-4 text-slate-400 transition-transform',
   450→                billExpanded && 'rotate-180',
   451→              )}
   452→            />
   453→          </button>
   454→          <AnimatePresence>
   455→            {billExpanded && (
   456→              <motion.div
   457→                initial={{ height: 0, opacity: 0 }}
   458→                animate={{ height: 'auto', opacity: 1 }}
   459→                exit={{ height: 0, opacity: 0 }}
   460→                transition={{ duration: 0.25 }}
   461→                className="overflow-hidden border-t border-slate-50 bg-slate-50/50"
   462→              >
   463→                <div className="space-y-1.5 px-4 py-3 text-[12.5px]">
   464→                  <BillRow label="Item total" value={order.subtotal} />
   465→                  {order.taxAmount > 0 && <BillRow label="Taxes & GST" value={order.taxAmount} />}
   466→                  {order.serviceCharge > 0 && (
   467→                    <BillRow label="Service charge" value={order.serviceCharge} />

   469→                  <div className="flex items-center justify-between border-t border-slate-200 pt-2">

   471→                    <span className="text-[14px] font-extrabold text-slate-900">
   472→                      {formatINR(order.grandTotal)}
   473→                    </span>
   474→                  </div>
   475→                </div>
   476→              </motion.div>
   477→            )}
   478→          </AnimatePresence>
   479→        </motion.div>
   480→
   481→        {/* ----- Status messages (paid / completed) ----- */}
   482→        {isPaid && !isCompleted && (
   483→          <motion.div
   484→            initial={{ y: 8, opacity: 0 }}
   485→            animate={{ y: 0, opacity: 1 }}
   486→            className="mt-3 flex items-center gap-2.5 rounded-xl bg-green-50 p-3.5"
   487→          >
   488→            <CheckCircle2 className="h-5 w-5 shrink-0 text-green-600" />
   489→            <p className="text-[13px] font-medium text-green-700">
   490→              Payment received — thank you!
   491→            </p>
   492→          </motion.div>
   493→        )}
   494→        {isCompleted && (
   495→          <motion.div
   496→            initial={{ y: 8, opacity: 0 }}
   497→            animate={{ y: 0, opacity: 1 }}
   498→            className="mt-3 flex items-center gap-2.5 rounded-xl bg-slate-50 p-3.5"
   499→          >
   500→            <Sparkles className="h-5 w-5 shrink-0 text-orange-500" />
   501→            <p className="text-[13px] font-medium text-slate-700">
   502→              Order completed. Thank you for dining with us!
   503→            </p>
   504→          </motion.div>
   505→        )}
   506→
   507→        {/* ----- Cancel (only on NEW / PENDING_PAYMENT orders) ----- */}
   508→        {canCancel && (
   509→          <div className="mt-3">
   510→            <Button
   511→              variant="ghost"
   512→              className="h-10 w-full rounded-xl text-[13px] font-medium text-red-500 hover:bg-red-50 hover:text-red-600"
   513→              disabled={cancel.isPending}
   514→              onClick={async () => {
   515→                try {
   516→                  await cancel.mutateAsync(orderId)
   517→                  toast.success('Order cancelled')
   518→                  onBackToMenu()
   519→                } catch (err: any) {
   520→                  toast.error(err.message || 'Could not cancel')
   521→                }
   522→              }}
   523→            >
   524→              <XCircle className="mr-2 h-4 w-4" />
   525→              Cancel order
   526→            </Button>
   527→          </div>
   528→        )}
   529→      </div>
   530→
   531→      {/* ----- Sticky bottom action bar ----- */}
   532→      <div className="fixed bottom-0 left-0 right-0 z-30 border-t border-slate-100 bg-white/95 backdrop-blur-md">
   533→        <div className="mx-auto max-w-2xl px-4 py-3 pb-[calc(0.75rem+env(safe-area-inset-bottom))]">
   534→          {needsToPay ? (
   535→            <Button
   536→              size="lg"
   537→              className="h-12 w-full rounded-xl bg-orange-600 text-[15px] font-bold text-white shadow-lg shadow-orange-600/30 transition-all hover:bg-orange-700 active:scale-[0.99]"
   538→              onClick={() => setCheckoutOpen(true)}
   539→            >
   540→              <CreditCard className="mr-2 h-4 w-4" />
   541→              Pay {formatINR(order.grandTotal)}
   542→              <ArrowRight className="ml-2 h-4 w-4" />
   543→            </Button>
   544→          ) : canRequestBill ? (
   545→            <Button
   546→              size="lg"
   547→              className="h-12 w-full rounded-xl bg-orange-600 text-[15px] font-bold text-white shadow-lg shadow-orange-600/30 transition-all hover:bg-orange-700 active:scale-[0.99]"
   548→              onClick={onProceedToBill}
   549→            >
   550→              Proceed to bill
   551→              <ArrowRight className="ml-2 h-4 w-4" />
   552→            </Button>
   553→          ) : isCompleted ? (
   554→            <Button
   555→              size="lg"
   556→              className="h-12 w-full rounded-xl bg-slate-900 text-[15px] font-bold text-white transition-all hover:bg-slate-800 active:scale-[0.99]"
   557→              onClick={onBackToMenu}
   558→            >
   559→              Place another order
   560→            </Button>
   561→          ) : (
   562→            <Button
   563→              size="lg"
   564→              variant="outline"
   565→              className="h-12 w-full rounded-xl border-slate-300 text-[15px] font-semibold text-slate-700 hover:bg-slate-50"
   566→              onClick={onBackToMenu}
   567→            >
   568→              Back to menu
   569→            </Button>
   570→          )}
   571→        </div>
   572→      </div>
   573→
   574→      {/* ----- Zepto-style Checkout Sheet ----- */}
   575→      <CheckoutSheet
   576→        open={checkoutOpen}
   577→        onOpenChange={setCheckoutOpen}
   578→        restaurant={restaurant as RestaurantInfo}
   579→        totals={{
   580→          itemCount: order.items?.length || 0,
   581→          subtotal: order.subtotal,
   582→          taxAmount: order.taxAmount,
   583→          serviceCharge: order.serviceCharge,
   584→          grandTotal: order.grandTotal,
   585→        }}
   586→        items={[]}
   587→        existingOrderId={order.id}
   588→        skipCustomerDetails
   589→        // Hide the pay-on-delivery method(s) the customer already chose at
   590→        // order placement — they shouldn't be asked to pick it again.
   591→        hiddenMethodIds={hiddenMethodIds}
   592→        onCheckoutComplete={() => {
   593→          setCheckoutOpen(false)
   594→          qc.invalidateQueries({ queryKey: ['customer-order', orderId] })
   595→        }}
   596→      />
   597→    </div>
   598→  )
   599→}
