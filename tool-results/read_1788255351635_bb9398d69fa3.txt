    66→    66→// RolePermission: per-role CRUD action permissions on resources (molecular level)
    67→    67→//   e.g. { role: MANAGER, resource: MENU_ITEM, action: CREATE, allowed: true }
    68→    68→// If no row exists for a (role, resource, action) tuple, the system falls back
    69→    69→// to the static PERMISSIONS map in src/lib/auth.ts. Once a row exists, it
    70→    70→// overrides the static default (allow OR deny).
    71→    71→model RolePermission {
    72→    72→  id          String   @id @default(cuid())
    73→    73→  role        String   // SUPER_ADMIN | RESTAURANT_OWNER | MANAGER | KITCHEN_STAFF | WAITER | CASHIER
    74→    74→  resource    String   // e.g. MENU_ITEM, ORDER, TABLE, STAFF, REPORTS, SETTINGS, BILLING, PLATFORM, etc.
    75→    75→  action      String   // CREATE | READ | UPDATE | DELETE (CRUD) or custom (VIEW, MANAGE, VERIFY, etc.)
    76→    76→  allowed     Boolean  @default(true)
    77→    77→  updatedBy   String?
    78→    78→  updatedByUser User?  @relation(fields: [updatedBy], references: [id])
    79→    79→
    80→    80→  createdAt   DateTime @default(now())
    81→    81→  updatedAt   DateTime @updatedAt
    82→    82→
    83→    83→  @@unique([role, resource, action])
    84→    84→  @@index([role])
    85→    85→  @@index([resource])
    86→    86→}

   233→   233→// RESTAURANT PAYMENT METHODS (Zepto-style configurable payment options)
   234→   234→// ============================================================================
   235→   235→// Each row = one payment option offered by a restaurant at checkout.
   236→   236→// A restaurant can configure label, description, icon, accent color, priority,
   237→   237→// and a JSON `config` blob for provider-specific settings (e.g. UPI VPA,
   238→   238→// Razorpay key ID, etc.). The `type` field drives the customer-facing flow.
   239→   239→//
   240→   240→// Type values:
   241→   241→//   UPI         → UPI deep-link + QR fallback
   242→   242→//   QR          → Standalone QR (renders a QR for the configured payload)
   243→   243→//   CARD        → Credit / Debit card (mock provider)
   244→   244→//   WALLET      → Mobile wallet (PhonePe / Paytm / Mobikwik etc.)
   245→   245→//   NETBANKING  → Net banking redirect (mock)
   246→   246→//   CASH        → Pay cash to waiter / counter staff
   247→   247→//   COUNTER     → Pay at counter (cash or card, in person)
   248→   248→//   PAY_LATER   → Pay-later / Tab (post-paid, settled at end of meal)
   249→   249→//   CUSTOM      → Restaurant-defined flow (renders as informational card)
   250→   250→model RestaurantPaymentMethod {
   251→   251→  id            String   @id @default(cuid())
   252→   252→  restaurantId  String
   253→   253→
   254→   254→  // Canonical type — drives the customer-facing payment flow.
   255→   255→  // See model comment above for the value set.
   256→   256→  type          String
   257→   257→
   258→   258→  // Display label shown to customers (e.g. "UPI", "PhonePe", "Pay at counter")
   259→   259→  label         String
   260→   260→  // Short helper text shown under the label (e.g. "GPay / PhonePe / Paytm")
   261→   261→  description   String?
   262→   262→  // Lucide icon name (e.g. "Smartphone", "CreditCard", "Banknote", "Wallet",
   263→   263→  // "Building", "QrCode", "Clock"). UI falls back to a default if missing.
   264→   264→  icon          String?
   265→   265→  // Tailwind-friendly hex color used for the icon chip (e.g. "#EA580C")
   266→   266→  accentColor   String   @default("#64748B")
   267→   267→
   268→   268→  // Whether this method is shown to customers at checkout
   269→   269→  active        Boolean  @default(true)
   270→   270→  // Sort order — lower numbers appear first in the customer picker
   271→   271→  priority      Int      @default(100)
   272→   272→
   273→   273→  // JSON blob with provider-specific config:
   274→   274→  //   UPI  → { upiId: "restaurant@okhdfcbank" }
   275→   275→  //   CARD → { provider: "RAZORPAY" | "STRIPE" | "MOCK",KeyId?: "...", keySecret?: "..." }
   276→   276→  //   WALLET → { provider: "PHONEPE" | "PAYTM" | "MOBIKWIK" | "MOCK" }
   277→   277→  //   QR → { payload: "upi://..." | override }
   278→   278→  config        Json?
   279→   279→
   280→   280→  restaurant    Restaurant @relation(fields: [restaurantId], references: [id], onDelete: Cascade)
   281→   281→
   282→   282→  createdAt DateTime @default(now())
   283→   283→  updatedAt DateTime @updatedAt
   284→   284→
   285→   285→  @@index([restaurantId])
   286→   286→  @@index([restaurantId, active])
   287→   287→}

   969→   969→// PLATFORM FEE PAYMENTS — end-to-end platform fee collection system
   970→   970→// ============================================================================
   971→   971→// When a restaurant pays its accumulated platform fees to the platform
   972→   972→// operator (super admin), a PlatformFeePayment row records the lump-sum
   973→   973→// payment. The payment covers a set of underlying PlatformFee rows (the
   974→   974→// per-order PENDING fees that the restaurant is settling).
   975→   975→//
   976→   976→// Lifecycle:
   977→   977→//   1. Super admin sends a PlatformFeePaymentRequest to a tenant with the
   978→   978→//      requested amount, due date, and note. Status = PENDING.
   979→   979→//   2. The tenant sees the request in their Settings → Platform Fees tab
   980→   980→//      and taps "Pay now" — they choose a payment method (UPI / Card /
   981→   981→//      Wallet / Net banking) and submit. A PlatformFeePayment row is
   982→   982→//      created with status = PROCESSING.
   983→   983→//   3. The platform verifies the payment (mock for UPI/CARD/etc.; real
   984→   984→//      gateway integration point) and the PlatformFeePayment transitions
   985→   985→//      to PAID. The covered PlatformFee rows are marked COLLECTED in bulk.
   986→   986→//   4. Realtime events fire at each step:
   987→   987→//        platform:feeRequested   → step 1 (admin → tenant)
   988→   988→//        platform:feePaymentInit → step 2 (tenant → admin)
   989→   989→//        platform:feePaid        → step 3 (system → both parties)
   990→   990→//        platform:feeCollected  → step 3 (system → both parties, per fee)
   991→   991→//
   992→   992→// Super admin can also do a "manual collect" (mark fees as collected
   993→   993→// without a payment) for offline settlements — status = COLLECTED_DIRECT.
   994→   994→
   995→   995→model PlatformFeePayment {
   996→   996→  id              String   @id @default(cuid())
   997→   997→  restaurantId    String
   998→   998→  // Amount in rupees (lump sum)
   999→   999→  amount          Float
  1000→  1000→  // Method: UPI | QR | CARD | WALLET | NETBANKING | CASH | MANUAL
  1001→  1001→  method          String
  1002→  1002→  // Status: PENDING | PROCESSING | PAID | FAILED | REFUNDED | COLLECTED_DIRECT
  1003→  1003→  status          String   @default("PENDING")
  1004→  1004→  // Provider references (for online payments)
  1005→  1005→  provider        String?  // UPI | MOCK | RAZORPAY | PHONEPE | ...
  1006→  1006→  providerTxnId   String?
  1007→  1007→  providerRef     String?
  1008→  1008→  idempotencyKey  String?  @unique
  1009→  1009→  // Verification
  1010→  1010→  verifiedAt      DateTime?
  1011→  1011→  verifiedById    String?
  1012→  1012→  failureReason   String?
  1013→  1013→
  1014→  1014→  // Links to a payment request (if the tenant is paying off a specific
  1015→  1015→  // request from the super admin). Null if this is a self-initiated payment.
  1016→  1016→  requestId       String?
  1017→  1017→
  1018→  1018→  // Notes
  1019→  1019→  note            String?
  1020→  1020→
  1021→  1021→  // Metadata
  1022→  1022→  rawResponse     String?
  1023→  1023→
  1024→  1024→  restaurant      Restaurant @relation(fields: [restaurantId], references: [id], onDelete: Cascade)
  1025→  1025→  verifiedBy      User?       @relation("PlatformFeePaymentVerifiedBy", fields: [verifiedById], references: [id])
  1026→  1026→  request         PlatformFeePaymentRequest? @relation(fields: [requestId], references: [id], onDelete: SetNull)
  1027→  1027→  coveredFees     PlatformFeeCoveredEntry[]
  1028→  1028→
  1029→  1029→  createdAt DateTime @default(now())
  1030→  1030→  updatedAt DateTime @updatedAt
  1031→  1031→
  1032→  1032→  @@index([restaurantId])
  1033→  1033→  @@index([status])
  1034→  1034→  @@index([createdAt])
  1035→  1035→}
