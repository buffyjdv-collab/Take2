    66→// RolePermission: per-role CRUD action permissions on resources (molecular level)
    67→//   e.g. { role: MANAGER, resource: MENU_ITEM, action: CREATE, allowed: true }
    68→// If no row exists for a (role, resource, action) tuple, the system falls back
    69→// to the static PERMISSIONS map in src/lib/auth.ts. Once a row exists, it
    70→// overrides the static default (allow OR deny).
    71→model RolePermission {
    72→  id          String   @id @default(cuid())
    73→  role        String   // SUPER_ADMIN | RESTAURANT_OWNER | MANAGER | KITCHEN_STAFF | WAITER | CASHIER
    74→  resource    String   // e.g. MENU_ITEM, ORDER, TABLE, STAFF, REPORTS, SETTINGS, BILLING, PLATFORM, etc.
    75→  action      String   // CREATE | READ | UPDATE | DELETE (CRUD) or custom (VIEW, MANAGE, VERIFY, etc.)
    76→  allowed     Boolean  @default(true)
    77→  updatedBy   String?
    78→  updatedByUser User?  @relation(fields: [updatedBy], references: [id])
    79→
    80→  createdAt   DateTime @default(now())
    81→  updatedAt   DateTime @updatedAt
    82→
    83→  @@unique([role, resource, action])
    84→  @@index([role])
    85→  @@index([resource])
    86→}

   232→// ============================================================================
   233→// RESTAURANT PAYMENT METHODS (Zepto-style configurable payment options)
   234→// ============================================================================
   235→// Each row = one payment option offered by a restaurant at checkout.
   236→// A restaurant can configure label, description, icon, accent color, priority,
   237→// and a JSON `config` blob for provider-specific settings (e.g. UPI VPA,
   238→// Razorpay key ID, etc.). The `type` field drives the customer-facing flow.
   239→//
   240→// Type values:
   241→//   UPI         → UPI deep-link + QR fallback
   242→//   QR          → Standalone QR (renders a QR for the configured payload)
   243→//   CARD        → Credit / Debit card (mock provider)
   244→//   WALLET      → Mobile wallet (PhonePe / Paytm / Mobikwik etc.)
   245→//   NETBANKING  → Net banking redirect (mock)
   246→//   CASH        → Pay cash to waiter / counter staff
   247→//   COUNTER     → Pay at counter (cash or card, in person)
   248→//   PAY_LATER   → Pay-later / Tab (post-paid, settled at end of meal)
   249→//   CUSTOM      → Restaurant-defined flow (renders as informational card)
   250→model RestaurantPaymentMethod {
   251→  id            String   @id @default(cuid())
   252→  restaurantId  String
   253→
   254→  // Canonical type — drives the customer-facing payment flow.
   255→  // See model comment above for the value set.
   256→  type          String
   257→
   258→  // Display label shown to customers (e.g. "UPI", "PhonePe", "Pay at counter")
   259→  label         String
   260→  // Short helper text shown under the label (e.g. "GPay / PhonePe / Paytm")
   261→  description   String?
   262→  // Lucide icon name (e.g. "Smartphone", "CreditCard", "Banknote", "Wallet",
   263→  // "Building", "QrCode", "Clock"). UI falls back to a default if missing.
   264→  icon          String?
   265→  // Tailwind-friendly hex color used for the icon chip (e.g. "#EA580C")
   266→  accentColor   String   @default("#64748B")
   267→
   268→  // Whether this method is shown to customers at checkout
   269→  active        Boolean  @default(true)
   270→  // Sort order — lower numbers appear first in the customer picker
   271→  priority      Int      @default(100)
   272→
   273→  // JSON blob with provider-specific config:
   274→  //   UPI  → { upiId: "restaurant@okhdfcbank" }
   275→  //   CARD → { provider: "RAZORPAY" | "STRIPE" | "MOCK",KeyId?: "...", keySecret?: "..." }
   276→  //   WALLET → { provider: "PHONEPE" | "PAYTM" | "MOBIKWIK" | "MOCK" }
   277→  //   QR → { payload: "upi://..." | override }
   278→  config        Json?
   279→
   280→  restaurant    Restaurant @relation(fields: [restaurantId], references: [id], onDelete: Cascade)
   281→
   282→  createdAt DateTime @default(now())
   283→  updatedAt DateTime @updatedAt
   284→
   285→  @@index([restaurantId])
   286→  @@index([restaurantId, active])
   287→}

   483→model Order {
   484→  id            String   @id @default(cuid())
   485→  orderNumber   String   // human-friendly: SG-000001
   486→  restaurantId  String
   487→  branchId      String?
   488→  tableId       String
   489→  customerId    String?
   490→  // Status: PENDING_PAYMENT | NEW | ACCEPTED | PREPARING | READY | SERVED | COMPLETED | CANCELLED
   491→  // PENDING_PAYMENT = customer placed the order but hasn't paid yet (pre-payment mode).
   492→  //   The order is hidden from the kitchen/orders module until payment is confirmed,
   493→  //   at which point it transitions to NEW and becomes visible for acceptance.
   494→  status        String   @default("NEW")
   495→  // Payment status: PENDING | PROCESSING | PAID | FAILED | REFUNDED
   496→  paymentStatus String   @default("PENDING")
   497→  // Payment method: UPI | CARD | CASH | COUNTER | WALLET
   498→  paymentMethod String?
   499→  // ID of the RestaurantPaymentMethod row the customer chose at order placement.
   500→  // Used by the post-serve payment picker to hide the same option (e.g. "Pay in cash"
   501→  // already chosen — don't show it again). Null for legacy orders / when no method
   502→  // was recorded at placement time.
   503→  paymentMethodId String?
   504→
   505→  // Special instructions from customer
   506→  notes         String?
   507→  // Idempotency key to prevent duplicate order submission
   508→  idempotencyKey String? @unique
   509→
   510→  // Customer snapshot (captured at order placement so the restaurant always
   511→  // has the name & phone even if the Customer record is later deleted)
   512→  customerName  String?
   513→  customerPhone String?
   514→
   515→  // Payment timing flags (set by restaurant owner)
   516→  // prePaymentRequested: restaurant asks customer to pay BEFORE the order is accepted
   517→  // postPaymentRequested: restaurant asks customer to pay AFTER the order is received (served)
   518→  prePaymentRequested  Boolean  @default(false)
   519→  postPaymentRequested Boolean  @default(false)
   520→  prePaymentRequestedAt  DateTime?
   521→  postPaymentRequestedAt DateTime?
   522→
   523→  // Who marked a CASH payment as received (waiter / owner / manager / cashier).
   524→  // When a cash payment is collected, the user's id + name snapshot are stored
   525→  // so the payments report can attribute the collection to a specific person.
   526→  cashReceivedById   String?
   527→  cashReceivedByName String?
   528→  cashReceivedAt     DateTime?
   529→
   530→  // Pricing (server-validated)
   531→  subtotal      Float    @default(0)
   532→  taxAmount     Float    @default(0)
   533→  serviceCharge Float    @default(0)
   534→  discountAmount Float   @default(0)
   535→  discountCode  String?
   536→  refundAmount  Float    @default(0)   // amount refunded (partial or full)
   537→  refundedAt    DateTime?
   538→  refundReason  String?
   539→  grandTotal    Float    @default(0)   // original grand total (pre-refund)
   540→  netTotal      Float    @default(0)   // grandTotal - refundAmount (actual collected)
   541→
   542→  // Platform fee snapshot (stored at order time for audit)
   543→  platformFeeAmount Float @default(0)
   544→  platformFeeBase   Float @default(0)   // the base the fee was calculated on
   545→  platformFeePayer  String? // RESTAURANT | CUSTOMER | SPLIT
   546→
   547→  // Staff who handled the order
   548→  acceptedById  String?
   549→  servedById    String?
   550→
   551→  // Timing
   552→  placedAt      DateTime @default(now())
   553→  acceptedAt    DateTime?
   554→  preparingAt   DateTime?
   555→  readyAt       DateTime?
   556→  servedAt      DateTime?
   557→  completedAt   DateTime?
   558→  cancelledAt   DateTime?
   559→
   560→  restaurant    Restaurant @relation(fields: [restaurantId], references: [id], onDelete: Cascade)
   561→  branch        Branch?    @relation(fields: [branchId],   references: [id], onDelete: SetNull)
   562→  table         Table      @relation(fields: [tableId],    references: [id], onDelete: Cascade)
   563→  customer      Customer?  @relation(fields: [customerId], references: [id], onDelete: SetNull)

   968→// ============================================================================
   969→// PLATFORM FEE PAYMENTS — end-to-end platform fee collection system
   970→// ============================================================================
   971→// When a restaurant pays its accumulated platform fees to the platform
   972→// operator (super admin), a PlatformFeePayment row records the lump-sum
   973→// payment. The payment covers a set of underlying PlatformFee rows (the
   974→// per-order PENDING fees that the restaurant is settling).
   975→//
   976→// Lifecycle:
   977→//   1. Super admin sends a PlatformFeePaymentRequest to a tenant with the
   978→//      requested amount, due date, and note. Status = PENDING.
   979→//   2. The tenant sees the request in their Settings → Platform Fees tab
   980→//      and taps "Pay now" — they choose a payment method (UPI / Card /
   981→//      Wallet / Net banking) and submit. A PlatformFeePayment row is
   982→//      created with status = PROCESSING.
   983→//   3. The platform verifies the payment (mock for UPI/CARD/etc.; real
   984→//      gateway integration point) and the PlatformFeePayment transitions
   985→//      to PAID. The covered PlatformFee rows are marked COLLECTED in bulk.
   986→//   4. Realtime events fire at each step:
   987→//        platform:feeRequested   → step 1 (admin → tenant)
   988→//        platform:feePaymentInit → step 2 (tenant → admin)
   989→//        platform:feePaid        → step 3 (system → both parties)
   990→//        platform:feeCollected  → step 3 (system → both parties, per fee)
   991→//
   992→// Super admin can also do a "manual collect" (mark fees as collected
   993→// without a payment) for offline settlements — status = COLLECTED_DIRECT.
   994→
   995→model PlatformFeePayment {
   996→  id              String   @id @default(cuid())
   997→  restaurantId    String
   998→  // Amount in rupees (lump sum)
   999→  amount          Float
  1000→  // Method: UPI | QR | CARD | WALLET | NETBANKING | CASH | MANUAL
  1001→  method          String
  1002→  // Status: PENDING | PROCESSING | PAID | FAILED | REFUNDED | COLLECTED_DIRECT
  1003→  status          String   @default("PENDING")
  1004→  // Provider references (for online payments)
  1005→  provider        String?  // UPI | MOCK | RAZORPAY | PHONEPE | ...
  1006→  providerTxnId   String?
  1007→  providerRef     String?
  1008→  idempotencyKey  String?  @unique
  1009→  // Verification
  1010→  verifiedAt      DateTime?
  1011→  verifiedById    String?
  1012→  failureReason   String?
  1013→
  1014→  // Links to a payment request (if the tenant is paying off a specific
  1015→  // request from the super admin). Null if this is a self-initiated payment.
  1016→  requestId       String?
  1017→
  1018→  // Notes
  1019→  note            String?
  1020→
  1021→  // Metadata
  1022→  rawResponse     String?
  1023→
  1024→  restaurant      Restaurant @relation(fields: [restaurantId], references: [id], onDelete: Cascade)
  1025→  verifiedBy      User?       @relation("PlatformFeePaymentVerifiedBy", fields: [verifiedById], references: [id])
  1026→  request         PlatformFeePaymentRequest? @relation(fields: [requestId], references: [id], onDelete: SetNull)
  1027→  coveredFees     PlatformFeeCoveredEntry[]
  1028→
  1029→  createdAt DateTime @default(now())
  1030→  updatedAt DateTime @updatedAt
  1031→
  1032→  @@index([restaurantId])
  1033→  @@index([status])
  1034→  @@index([createdAt])
  1035→}
