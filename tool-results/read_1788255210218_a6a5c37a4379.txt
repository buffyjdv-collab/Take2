    67→    67→const STEPS: StepConfig[] = [
    68→    68→  { key: 'NEW', label: 'Order Placed', shortLabel: 'Placed', sublabel: 'Waiting for restaurant to accept', icon: CheckCircle2, etaText: '~20-25 min' },
    69→    69→  { key: 'ACCEPTED', label: 'Confirmed', shortLabel: 'Confirmed', sublabel: 'Restaurant has accepted your order', icon: BellRing, etaText: '~15-20 min' },
    70→    70→  { key: 'PREPARING', label: 'Preparing', shortLabel: 'Cooking', sublabel: 'The chef is preparing your food', icon: ChefHat, etaText: '~10-15 min' },
    71→    71→  { key: 'READY', label: 'Ready to Serve', shortLabel: 'Ready', sublabel: 'Your food is ready for serving', icon: PackageCheck, etaText: '~2-5 min' },
    72→    72→  { key: 'SERVED', label: 'Served', shortLabel: 'Served', sublabel: 'Enjoy your meal!', icon: Utensils, etaText: 'Enjoy!' },
    73→    73→]
    74→    74→
    75→    75→function stepIndex(status: string): number {
    76→    76→  const order: string[] = ['NEW', 'ACCEPTED', 'PREPARING', 'READY', 'SERVED', 'COMPLETED']
    77→    77→  const i = order.indexOf(status)
    78→    78→  if (i < 0) return -1
    79→    79→  return Math.min(i, STEPS.length - 1)
    80→    80→}

   411→   411→                  <VegBadge isVeg={it.isVeg} />
   412→   412→                </div>
   413→   413→                <div className="min-w-0 flex-1">
   414→   414→                  <p className="text-[13.5px] font-semibold text-slate-800">
   415→   415→                    {it.menuItemName}
   416→   416→                  </p>
   417→   417→                  {it.variantName && (
   418→   418→                    <p className="mt-0.5 text-[11.5px] text-slate-400">{it.variantName}</p>
   419→   419→                  )}
   420→   420→                  {it.modifiers?.length > 0 && (
   421→   421→                    <p className="mt-0.5 truncate text-[11.5px] text-slate-400">
   422→   422→                      + {it.modifiers.map((m: any) => m.modifierName).join(', ')}
   423→   423→                    </p>
   424→   424→                  )}
   425→   425→                </div>
   426→   426→                <div className="shrink-0 text-right">
   427→   427→                  <p className="text-[12px] text-slate-400">
   428→   428→                    {it.quantity} × {formatINR(it.unitPrice || it.totalPrice / it.quantity)}
   429→   429→                  </p>
   430→   430→                  <p className="text-[13.5px] font-bold text-slate-900">
   431→   431→                    {formatINR(it.totalPrice)}
   432→   432→                  </p>
   433→   433→                </div>
   434→   434→              </div>
   435→   435→            ))}
   436→   436→          </div>
   437→   437→
   438→   438→          {/* Bill details — collapsible */}
   439→   439→          <button
   440→   440→            onClick={() => setBillExpanded((v) => !v)}
   441→   441→            className="flex w-full items-center justify-between border-t border-slate-100 px-4 py-3 text-[13px] font-semibold text-slate-700 transition-colors hover:bg-slate-50"
   442→   442→          >
   443→   443→            <span className="flex items-center gap-2">
   444→   444→              <Receipt className="h-4 w-4 text-slate-400" />
   445→   445→              Bill details
   446→   446→            </span>
   447→   447→            <ChevronDown
   448→   448→              className={cn(
   449→   449→                'h-4 w-4 text-slate-400 transition-transform',
   450→   450→                billExpanded && 'rotate-180',
   451→   451→              )}
   452→   452→            />
   453→   453→          </button>
   454→   454→          <AnimatePresence>
   455→   455→            {billExpanded && (
   456→   456→              <motion.div
   457→   457→                initial={{ height: 0, opacity: 0 }}
   458→   458→                animate={{ height: 'auto', opacity: 1 }}
   459→   459→                exit={{ height: 0, opacity: 0 }}
   460→   460→                transition={{ duration: 0.25 }}
   461→   461→                className="overflow-hidden border-t border-slate-50 bg-slate-50/50"
   462→   462→              >
   463→   463→                <div className="space-y-1.5 px-4 py-3 text-[12.5px]">
   464→   464→                  <BillRow label="Item total" value={order.subtotal} />
   465→   465→                  {order.taxAmount > 0 && <BillRow label="Taxes & GST" value={order.taxAmount} />}
   466→   466→                  {order.serviceCharge > 0 && (
   467→   467→                    <BillRow label="Service charge" value={order.serviceCharge} />
   468→   468→                  )}
   469→   469→                  <div className="flex items-center justify-between border-t border-slate-200 pt-2">
   470→   470→                    <span className="text-[13.5px] font-extrabold text-slate-900">Grand total</span>
   471→   471→                    <span className="text-[14px] font-extrabold text-slate-900">
   472→   472→                      {formatINR(order.grandTotal)}
   473→   473→                    </span>
   474→   474→                  </div>
   475→   475→                </div>
   476→   476→              </motion.div>
   477→   477→            )}
   478→   478→          </AnimatePresence>
   479→   479→        </motion.div>
   480→   480→
   481→   481→        {/* ----- Status messages (paid / completed) ----- */}
   482→   482→        {isPaid && !isCompleted && (
   483→   483→          <motion.div
   484→   484→            initial={{ y: 8, opacity: 0 }}
   485→   485→            animate={{ y: 0, opacity: 1 }}
   486→   486→            className="mt-3 flex items-center gap-2.5 rounded-xl bg-green-50 p-3.5"
   487→   487→          >
   488→   488→            <CheckCircle2 className="h-5 w-5 shrink-0 text-green-600" />
   489→   489→            <p className="text-[13px] font-medium text-green-700">
   490→   490→              Payment received — thank you!
   491→   491→            </p>
   492→   492→          </motion.div>
   493→   493→        )}
   494→   494→        {isCompleted && (
   495→   495→          <motion.div
   496→   496→            initial={{ y: 8, opacity: 0 }}
   497→   497→            animate={{ y: 0, opacity: 1 }}
   498→   498→            className="mt-3 flex items-center gap-2.5 rounded-xl bg-slate-50 p-3.5"
   499→   499→          >
   500→   500→            <Sparkles className="h-5 w-5 shrink-0 text-orange-500" />
   501→   501→            <p className="text-[13px] font-medium text-slate-700">
   502→   502→              Order completed. Thank you for dining with us!
   503→   503→            </p>
   504→   504→          </motion.div>
   505→   505→        )}
   506→   506→
   507→   507→        {/* ----- Cancel (only on NEW / PENDING_PAYMENT orders) ----- */}
   508→   508→        {canCancel && (
   509→   509→          <div className="mt-3">
   510→   510→            <Button
   511→   511→              variant="ghost"
   512→   512→              className="h-10 w-full rounded-xl text-[13px] font-medium text-red-500 hover:bg-red-50 hover:text-red-600"
   513→   513→              disabled={cancel.isPending}
   514→   514→              onClick={async () => {
   515→   515→                try {
   516→   516→                  await cancel.mutateAsync(orderId)
   517→   517→                  toast.success('Order cancelled')
   518→   518→                  onBackToMenu()
   519→   519→                } catch (err: any) {
   520→   520→                  toast.error(err.message || 'Could not cancel')
   521→   521→                }
   522→   522→              }}
   523→   523→            >
   524→   524→              <XCircle className="mr-2 h-4 w-4" />
   525→   525→              Cancel order
   526→   526→            </Button>
   527→   527→          </div>
   528→   528→        )}
   529→   529→      </div>
   530→   530→
   531→   531→      {/* ----- Sticky bottom action bar ----- */}
   532→   532→      <div className="fixed bottom-0 left-0 right-0 z-30 border-t border-slate-100 bg-white/95 backdrop-blur-md">
   533→   533→        <div className="mx-auto max-w-2xl px-4 py-3 pb-[calc(0.75rem+env(safe-area-inset-bottom))]">
   534→   534→          {needsToPay ? (
   535→   535→            <Button
   536→   536→              size="lg"
   537→   537→              className="h-12 w-full rounded-xl bg-orange-600 text-[15px] font-bold text-white shadow-lg shadow-orange-600/30 transition-all hover:bg-orange-700 active:scale-[0.99]"
   538→   538→              onClick={() => setCheckoutOpen(true)}
   539→   539→            >
   540→   540→              <CreditCard className="mr-2 h-4 w-4" />
   541→   541→              Pay {formatINR(order.grandTotal)}
   542→   542→              <ArrowRight className="ml-2 h-4 w-4" />
   543→   543→            </Button>
   544→   544→          ) : canRequestBill ? (
   545→   545→            <Button
   546→   546→              size="lg"
   547→   547→              className="h-12 w-full rounded-xl bg-orange-600 text-[15px] font-bold text-white shadow-lg shadow-orange-600/30 transition-all hover:bg-orange-700 active:scale-[0.99]"
   548→   548→              onClick={onProceedToBill}
   549→   549→            >
   550→   550→              Proceed to bill
   551→   551→              <ArrowRight className="ml-2 h-4 w-4" />
   552→   552→            </Button>
   553→   553→          ) : isCompleted ? (
   554→   554→            <Button
   555→   555→              size="lg"
   556→   556→              className="h-12 w-full rounded-xl bg-slate-900 text-[15px] font-bold text-white transition-all hover:bg-slate-800 active:scale-[0.99]"
   557→   557→              onClick={onBackToMenu}
   558→   558→            >
   559→   559→              Place another order
   560→   560→            </Button>
   561→   561→          ) : (
   562→   562→            <Button
   563→   563→              size="lg"
   564→   564→              variant="outline"
   565→   565→              className="h-12 w-full rounded-xl border-slate-300 text-[15px] font-semibold text-slate-700 hover:bg-slate-50"
   566→   566→              onClick={onBackToMenu}
   567→   567→            >
   568→   568→              Back to menu
   569→   569→            </Button>
   570→   570→          )}
   571→   571→        </div>
   572→   572→      </div>
   573→   573→
   574→   574→      {/* ----- Zepto-style Checkout Sheet ----- */}
   575→   575→      <CheckoutSheet
   576→   576→        open={checkoutOpen}
   577→   577→        onOpenChange={setCheckoutOpen}
   578→   578→        restaurant={restaurant as RestaurantInfo}
   579→   579→        totals={{
   580→   580→          itemCount: order.items?.length || 0,
   581→   581→          subtotal: order.subtotal,
   582→   582→          taxAmount: order.taxAmount,
   583→   583→          serviceCharge: order.serviceCharge,
   584→   584→          grandTotal: order.grandTotal,
   585→   585→        }}
   586→   586→        items={[]}
   587→   587→        existingOrderId={order.id}
   588→   588→        skipCustomerDetails
   589→   589→        // Hide the pay-on-delivery method(s) the customer already chose at
   590→   590→        // order placement — they shouldn't be asked to pick it again.
   591→   591→        hiddenMethodIds={hiddenMethodIds}
   592→   592→        onCheckoutComplete={() => {
   593→   593→          setCheckoutOpen(false)
   594→   594→          qc.invalidateQueries({ queryKey: ['customer-order', orderId] })
   595→   595→        }}
   596→   596→      />
   597→   597→    </div>
   598→   598→  )
   599→   599→}
